cmake_minimum_required(VERSION 3.22)
project(qnn_runtime VERSION 0.1.0 LANGUAGES CXX)
find_package(OpenCV REQUIRED COMPONENTS core)
set(QNN_INCLUDE_DIR "" CACHE PATH "Compatible QNN C API headers")
if(NOT EXISTS "${QNN_INCLUDE_DIR}/QnnInterface.h")
  message(FATAL_ERROR "Set QNN_INCLUDE_DIR to the QNN include/QNN directory")
endif()
add_library(qnn_runtime STATIC src/session.cpp)
target_compile_features(qnn_runtime PUBLIC cxx_std_20)
target_include_directories(qnn_runtime PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_include_directories(qnn_runtime SYSTEM PRIVATE "${QNN_INCLUDE_DIR}")
target_link_libraries(qnn_runtime PUBLIC ${OpenCV_LIBRARIES} ${CMAKE_DL_LIBS})
target_compile_options(qnn_runtime PRIVATE -Wall -Wextra -Wpedantic -Werror)

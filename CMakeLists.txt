cmake_minimum_required(VERSION 3.22)
project(dst_qcs8550 VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(MID_WORKSPACE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
include(CTest)
option(DST_BUILD_RBT "Build the optional RBT capture/window component" ON)
option(DST_ENABLE_QNN "Build the native QNN evaluator adapter" OFF)
if(DST_BUILD_RBT)
    add_subdirectory(src/dst_calib_rbt)
elseif(DST_ENABLE_QNN)
    add_subdirectory(src/qnn_runtime)
    add_library(dst_qnn SHARED src/dst_calib_rbt/src/qnn_evaluator.cpp)
    target_compile_features(dst_qnn PRIVATE cxx_std_20)
    target_link_libraries(dst_qnn PRIVATE qnn_runtime)
    set_target_properties(dst_qnn PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/dst-calib/lib")
endif()
